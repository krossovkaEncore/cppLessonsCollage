### **1. Перевести год в наносекунды**

```
#include <iostream>
using namespace std;

int main() {
    long long seconds = 31536000;

    cout << seconds * 1000000000LL << " наносекунд";

    return 0;
}
```

### **2. Найти гравитационную силу**

```
#include <iostream>
#include <cmath>
using namespace std;

int main() {
    double G, m1, m2, r;

    cin >> G >> m1 >> m2 >> r;

    double F = G * m1 * m2 / pow(r, 2);

    cout << "F = " << F << endl;

    return 0;
}
```
