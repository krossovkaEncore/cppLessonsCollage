### **1. Вывести большее из двух чисел**

```
#include <iostream>
using namespace std;

int main() {
    int a, b;
    cin >> a >> b;

    if (a > b)
        cout << a;
    else
        cout << b;

    return 0;
}
```

### **2. Сортировка трёх чисел по возрастанию**

```
#include <iostream>
using namespace std;

int main() {
    int a, b, c;
    cin >> a >> b >> c;

    if (a > b) swap(a, b);
    if (a > c) swap(a, c);
    if (b > c) swap(b, c);

    cout << a << " " << b << " " << c;

    return 0;
}
```
