### **1. Калькулятор с помощью тернарного оператора**

```
#include <iostream>
using namespace std;

int main() {
    double a, b;
    char op;

    cin >> a >> op >> b;

    cout << (op == '+' ? a + b :
             op == '-' ? a - b :
             op == '*' ? a * b :
             op == '/' ? a / b : 0);

    return 0;
}
```

### **2. Является ли год високосным**

```
#include <iostream>
using namespace std;

int main() {
    int year;
    cin >> year;

    cout << ((year % 400 == 0 || (year % 4 == 0 && year % 100 != 0))
             ? "Високосный" : "Не високосный");

    return 0;
}
```
