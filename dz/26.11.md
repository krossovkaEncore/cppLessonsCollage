### **1. Суммировать числа, пока не введён 0**

```
#include <iostream>
using namespace std;

int main() {
    int n, sum = 0;

    do {
        cin >> n;
        sum += n;
    } while (n != 0);

    cout << "Сумма: " << sum;

    return 0;
}
```

### **2. Запрашивать пароль, пока он не будет правильным**

```
#include <iostream>
#include <string>
using namespace std;

int main() {
    string password = "12345";
    string input;

    do {
        cout << "Введите пароль: ";
        cin >> input;
    } while (input != password);

    cout << "Пароль верный!";

    return 0;
}
```
