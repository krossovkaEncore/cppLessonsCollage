### **1. Вывести большее из двух чисел**

```
#include <iostream>
using namespace std;

int main() {
    double a, b;
    cin >> a >> b;

    if (a > b)
        cout << a;
    else
        cout << b;

    return 0;
}
```

### **2. Вывести число, только если оно входит в диапазон (0; 100)**

```
#include <iostream>
using namespace std;

int main() {
    double a;
    cin >> a;

    if (a > 0 && a < 100)
        cout << a;

    return 0;
}
```
