### **1. Вывести простые числа от 0 до 10000**

```
#include <iostream>
using namespace std;

int main() {
    for (int i = 2; i <= 10000; i++) {
        bool prime = true;

        for (int j = 2; j * j <= i; j++) {
            if (i % j == 0) {
                prime = false;
                break;
            }
        }

        if (prime)
            cout << i << " ";
    }

    return 0;
}
```

### **2. Игра «Угадай число»**

```
#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

int main() {
    srand(time(0));
    int number = rand() % 101;
    int guess;

    while (true) {
        cin >> guess;

        if (guess == number) {
            cout << "Угадал!";
            break;
        }

        continue;
    }

    return 0;
}
```

**Интересный факт:** в первой программе проверка `j * j <= i` позволяет не проверять делители дальше квадратного корня числа. Это заметно ускоряет поиск простых чисел.
