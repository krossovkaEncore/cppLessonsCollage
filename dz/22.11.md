### **1. Выбор писателя и вывод популярной книги**

```
#include <iostream>
using namespace std;

int main() {
    int choice;
    cout << "1. Пушкин\n2. Достоевский\n3. Толстой\n4. Чехов\n";
    cin >> choice;

    switch (choice) {
        case 1: cout << "Евгений Онегин"; break;
        case 2: cout << "Преступление и наказание"; break;
        case 3: cout << "Война и мир"; break;
        case 4: cout << "Вишневый сад"; break;
        default: cout << "Неверный выбор";
    }

    return 0;
}
```

### **2. Выбор математической функции**

```
#include <iostream>
#include <cmath>
using namespace std;

int main() {
    int choice;
    double x, y;

    cout << "1. log\n2. pow\n3. sqrt\n4. cos\n5. sin\n";
    cin >> choice;

    switch (choice) {
        case 1:
            cin >> x;
            cout << log(x);
            break;
        case 2:
            cin >> x >> y;
            cout << pow(x, y);
            break;
        case 3:
            cin >> x;
            cout << sqrt(x);
            break;
        case 4:
            cin >> x;
            cout << cos(x);
            break;
        case 5:
            cin >> x;
            cout << sin(x);
            break;
        default:
            cout << "Неверный выбор";
    }

    return 0;
}
```

*Примечание: `sin()` и `cos()` принимают угол в радианах.*
